-- Export vessels with availability (overwrite: DataSets/Derived/Stena/SampleScenario/vessels_with_availability.csv)
WITH op AS (
  SELECT VESSEL_ID,
         OPEN_POSITION_PORT_ID AS VESSEL_OPEN_PORT_ID,
         OPEN_POSITION_PORT_DATE AS VESSEL_AVAILABLE_UTC,
         ROW_NUMBER() OVER (PARTITION BY VESSEL_ID ORDER BY OPEN_POSITION_PORT_DATE) rn
  FROM STENATEST.TBL_VESSEL_OPEN_POSITION
  WHERE SCENARIO_ID = :SCENARIO_ID
)
SELECT
  v.VESSEL_ID,
  v.VESSEL_NAME,
  v.VESSEL_TYPE_ID,
  v.DWT_SUMMER,
  v.DRAFT_SUMMER,
  v.LOA,
  v.BEAM,
  op.VESSEL_OPEN_PORT_ID,
  op.VESSEL_AVAILABLE_UTC
FROM STENATEST.TBL_VESSEL v
JOIN STENATEST.TBL_SCENARIO_HAS_VESSEL shv
  ON shv.VESSEL_ID = v.VESSEL_ID
 AND shv.SCENARIO_ID = :SCENARIO_ID
LEFT JOIN op
  ON op.VESSEL_ID = v.VESSEL_ID
 AND op.rn = 1
ORDER BY v.VESSEL_ID;

-- Export voyages with laycans + economics (overwrite: DataSets/Derived/Stena/SampleScenario/unallocated_voyages_detailed.csv)
WITH scn_voy AS (
  SELECT DISTINCT VOYAGE_ID
  FROM STENATEST.TBL_SCENARIO_HAS_VOYAGE
  WHERE SCENARIO_ID = :SCENARIO_ID
),
cargo_laycan AS (
  SELECT
         VOYAGE_ID,
         MIN(LAYCAN_FROM) AS LAYCAN_START_UTC,
         MAX(LAYCAN_TO)   AS LAYCAN_END_UTC
  FROM STENATEST.TBL_CARGO
  WHERE VOYAGE_ID IN (SELECT VOYAGE_ID FROM scn_voy)
  GROUP BY VOYAGE_ID
),
rfc AS (
  SELECT
         REASON_FOR_CALL_ID,
         UPPER(TRIM(REASON_FOR_CALL)) AS RFC_CODE
  FROM STENATEST.TBL_REASON_FOR_CALL
),
pc_ranked AS (
  SELECT
         pc.VOYAGE_ID,
         pc.PORT_ID,
         pc.PORT_CALL_SEQUENCE,
         rfc.RFC_CODE,
         ROW_NUMBER() OVER (
           PARTITION BY pc.VOYAGE_ID, rfc.RFC_CODE
           ORDER BY pc.PORT_CALL_SEQUENCE ASC
         ) AS RN_FIRST
  FROM STENATEST.TBL_PORT_CALL pc
  LEFT JOIN rfc
    ON rfc.REASON_FOR_CALL_ID = pc.REASON_FOR_CALL_ID
  WHERE pc.VOYAGE_ID IN (SELECT VOYAGE_ID FROM scn_voy)
),
voyage_ports AS (
  SELECT
         VOYAGE_ID,
         MAX(CASE WHEN RFC_CODE = 'L' AND RN_FIRST = 1 THEN PORT_ID END) AS ORIGIN_PORT_ID_PC,
         MAX(CASE WHEN RFC_CODE = 'D' AND RN_FIRST = 1 THEN PORT_ID END) AS DEST_PORT_ID_PC
  FROM pc_ranked
  GROUP BY VOYAGE_ID
)
SELECT
  v.VOYAGE_ID,
  v.VESSEL_ID,
  v.VOYAGE_STATUS,
  v.ESTIMATED_VOYAGE_START_DATE,
  v.HAS_CANAL_PASSAGE,
  CAST(NULL AS NUMBER) AS IS_BALLAST,
  v.CANAL_COST,
  v.MILES_DIRECT,
  v.MILES_BALLAST,
  v.MILES_LOADED,
  v.DAYS_TOTAL_AT_SEA,
  v.DAYS_TOTAL_IN_PORT,
  v.DAYS_BALLAST,
  v.DAYS_LOADED,
  v.DAYS_WAITING,
  NVL(vp.ORIGIN_PORT_ID_PC, v.BALLAST_PORT_ID) AS ORIGIN_PORT_ID,
  NVL(vp.DEST_PORT_ID_PC,   v.RP_LOADED_ID)    AS DEST_PORT_ID,
  v.RP_BALLAST_ID,
  v.RP_LOADED_ID,
  CAST(NULL AS NUMBER)   AS TERMINAL_ID,
  c.LAYCAN_START_UTC,
  c.LAYCAN_END_UTC,
  -- Economics
  v.FREIGHT,
  v.GROSS_FREIGHT,
  v.NET_FREIGHT,
  v.FREIGHT_COST,
  v.PORT_COST,
  v.HANDLING_COST,
  v.CANAL_COST AS VOYAGE_CANAL_COST,
  v.FO_COST,
  v.DO_COST,
  v.VARIOUS_COST,
  v.VARIOUS_REVENUE,
  v.RUNNING_COST,
  v.RUNNING_COST_PER_DAY
FROM STENATEST.TBL_VOYAGE v
JOIN scn_voy s ON s.VOYAGE_ID = v.VOYAGE_ID
LEFT JOIN cargo_laycan c ON c.VOYAGE_ID = v.VOYAGE_ID
LEFT JOIN voyage_ports vp ON vp.VOYAGE_ID = v.VOYAGE_ID
ORDER BY v.VOYAGE_ID;
